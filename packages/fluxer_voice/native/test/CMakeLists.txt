cmake_minimum_required(VERSION 3.15)
project(fluxer_noise_filter_test LANGUAGES C CXX)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(FNF_SANITIZE_THREAD "" OFF)
option(FNF_SANITIZE_ADDRESS "" OFF)
if(FNF_SANITIZE_THREAD)
  add_compile_options(-fsanitize=thread -g)
  add_link_options(-fsanitize=thread)
endif()
if(FNF_SANITIZE_ADDRESS)
  add_compile_options(-fsanitize=address,undefined -g)
  add_link_options(-fsanitize=address,undefined)
endif()

include("${CMAKE_CURRENT_SOURCE_DIR}/../noise_filter.cmake")

enable_testing()
add_executable(noise_filter_test noise_filter_test.cc)
target_compile_features(noise_filter_test PRIVATE cxx_std_17)
target_link_libraries(noise_filter_test PRIVATE fluxer_noise_filter)
if(FNF_SANITIZE_THREAD OR FNF_SANITIZE_ADDRESS)
  target_compile_definitions(noise_filter_test PRIVATE FNF_NO_MALLOC_HOOK)
endif()
add_test(NAME noise_filter_test COMMAND noise_filter_test)
